      # Bootstrap the `deces` index on both engines with the
      # INSEE 10k mapping + bulk-load slice baked into this
      # image. The slice NDJSON carries `_index: "deces"` in
      # every action header, so artillery_bench targets the
      # same index name below.
      bootstrap_engine() {
        url="$1"; label="$2"
        # The scratch PVC persists across ndcg-gate / insee-bench
        # runs, so OpenSearch / Surch boot with whatever indices
        # the previous job left behind (e.g. `scifact`,
        # `.opensearch-*` security indices). Drop the working
        # set before re-creating it, otherwise the bulk POST
        # below can 400 against a partially-allocated `deces`
        # index in a YELLOW cluster.
        curl -fsS -o /dev/null -X DELETE "$url/deces" 2>/dev/null || true
        # OpenSearch only — wait for cluster to become at least
        # YELLOW (single-node has no replicas) before we ask it
        # to ingest a brand new index.
        curl -fsS -o /dev/null \
          "$url/_cluster/health?wait_for_status=yellow&timeout=30s" \
          2>/dev/null || true
        echo "bootstrap $label: PUT /deces"
        curl -fsS -X PUT "$url/deces" \
          -H 'Content-Type: application/json' \
          --data-binary @/usr/local/share/deces/mapping.json \
          > "$REPORTS_DIR/bootstrap-$label.put.json"
        # Wait for the freshly-PUT `deces` index to have its
        # primary shard active before we bulk-load against it.
        curl -fsS -o /dev/null \
          "$url/_cluster/health/deces?wait_for_status=yellow&wait_for_active_shards=1&timeout=30s" \
          2>/dev/null || true
        echo "bootstrap $label: POST /_bulk (10k docs)"
        gunzip -c /usr/local/share/deces/slice-10000.ndjson.gz \
          | curl -fsS -X POST "$url/_bulk" \
              -H 'Content-Type: application/x-ndjson' \
              --data-binary @- \
              > "$REPORTS_DIR/bootstrap-$label.bulk.json"
        # Refresh so newly indexed docs are searchable before
        # the load phase starts.
        curl -fsS -X POST "$url/deces/_refresh" \
          > "$REPORTS_DIR/bootstrap-$label.refresh.json"
      }
      bootstrap_engine "$SURCH_URL" surch
      bootstrap_engine "$OPENSEARCH_URL" opensearch
      # Warm up both engines so JVM JIT + page caches are hot
      # before artillery_bench measures cold-start outliers
      # (OpenSearch phase-1 max would otherwise spike above
      # SLO_ARTILLERY_MAX_MS = 500 ms).
      warmup_engine() {
        url="$1"; label="$2"
        echo "warmup $label: 50 match_all hits"
        for i in $(seq 1 50); do
          curl -fsS -o /dev/null -X POST "$url/deces/_search" \
            -H 'Content-Type: application/json' \
            -d '{"query":{"match_all":{}},"size":5}' \
            || true
        done
      }
      warmup_engine "$SURCH_URL" surch
      warmup_engine "$OPENSEARCH_URL" opensearch

      # bench_report in /usr/local/bin.

        --report "$REPORTS_DIR/art-os.json"

        --dir "$REPORTS_DIR" \
        --output "$REPORTS_DIR/summary.md"
      echo "BEGIN_SURCH_K8S_SUMMARY"
      cat "$REPORTS_DIR/summary.md"
      echo "END_SURCH_K8S_SUMMARY"
      echo "insee-bench complete."
